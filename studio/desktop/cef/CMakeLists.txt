cmake_minimum_required(VERSION 3.24)
# The CEF Linux SDK checks both C and C++ compiler flags.
project(CyDocStudioCEF LANGUAGES C CXX)
set(CYDOC_CYUI_SOURCE_DIR "" CACHE PATH "CyUI source checkout")
if(NOT EXISTS "${CYDOC_CYUI_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set CYDOC_CYUI_SOURCE_DIR to the CyUI source checkout")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CYUI_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
set(CYUI_BUILD_GALLERY OFF CACHE BOOL "" FORCE)
set(CYUI_MODULES Web CACHE STRING "" FORCE)
set(CYUI_BUILD_WEB_CEF ON CACHE BOOL "" FORCE)
add_subdirectory("${CYDOC_CYUI_SOURCE_DIR}" cyui)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set_target_properties(cyui_webcef_helper cyui_webcef_bridge PROPERTIES
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
endif()
