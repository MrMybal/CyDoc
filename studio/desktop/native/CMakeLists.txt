cmake_minimum_required(VERSION 3.24)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../package.json" studio_package)
string(JSON studio_version GET "${studio_package}" version)
project(CyDocStudioHost VERSION ${studio_version} LANGUAGES CXX)
set(CYDOC_CYMCP_SOURCE_DIR "" CACHE PATH "CyMCPSDK 0.7.5 source checkout")
if(NOT EXISTS "${CYDOC_CYMCP_SOURCE_DIR}/Documentation/DocAI.md")
  message(FATAL_ERROR "Set CYDOC_CYMCP_SOURCE_DIR to the CyMCPSDK source checkout")
endif()
set(CYMCP_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_CHAT_APP OFF CACHE BOOL "" FORCE)
set(CYMCP_BUILD_CHAT_UI ON CACHE BOOL "" FORCE)
set(CYMCP_BUILD_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory("${CYDOC_CYMCP_SOURCE_DIR}" sdk)
if(NOT CyMCPSDK_VERSION VERSION_EQUAL "0.7.5")
  # Subdirectory project versions are scoped; inspect the public version contract too.
  file(STRINGS "${CYDOC_CYMCP_SOURCE_DIR}/include/cymcp/core.hpp" sdk_version REGEX "version = .0.7.5.")
  if(NOT sdk_version)
    message(FATAL_ERROR "CyDoc Studio requires CyMCPSDK 0.7.5")
  endif()
endif()
set(backends "${CYMCP_CYUI_SOURCE_DIR}/third_party/imgui/backends")
add_executable(CyDocStudioHost main.cpp chat.cpp remotes.cpp "${backends}/imgui_impl_win32.cpp" "${backends}/imgui_impl_dx11.cpp")
target_compile_features(CyDocStudioHost PRIVATE cxx_std_20)
target_compile_definitions(CyDocStudioHost PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
target_compile_options(CyDocStudioHost PRIVATE /utf-8 /W4)
target_include_directories(CyDocStudioHost PRIVATE "${backends}")
target_link_libraries(CyDocStudioHost PRIVATE CyMCP::Core CyMCP::Http CyMCP::MCP CyMCP::AI CyMCP::ChatUI d3d11 dxgi d3dcompiler dwmapi shell32 ole32 comdlg32)
install(TARGETS CyDocStudioHost RUNTIME DESTINATION .)
